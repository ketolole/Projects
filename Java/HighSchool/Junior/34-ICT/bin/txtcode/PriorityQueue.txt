import java.util.NoSuchElementException;


public class PriorityQueue<E>
{
  private ArrayList<E> myArray;


  public PriorityQueue()
  {
    myArray = new ArrayList<E>();
    myArray.add(null);
  }


  public void add(E item)
  {
    myArray.add(item);
    heapUp(item);
  }


  private void heapUp(E item)
  {
    int index = myArray.size()-1;
    while (index > 1 && ((Comparable) myArray.get(index / 2)).compareTo((Comparable) myArray.get(index)) > 0)
    {
      myArray.set(index, myArray.get(index / 2));
      myArray.set(index / 2, item);
      index /= 2;
    }
  }


  public E remove()
  {
    if(isEmpty())
    {
      throw new NoSuchElementException("java.util.NoSuchElementException: null");
    }
    E root = myArray.get(1);
    if(myArray.size() > 2)
    {
    myArray.set(1, myArray.remove(myArray.size()-1));
    heapDown(myArray.get(1));
    }
    else
    {
      myArray.remove(1);
    }
    return root;


  }


  private void heapDown(E item)
  {
    int index = 1;
    while (index < myArray.size())
    {
      if (2 * index + 1 < myArray.size())
      {
        if (((Comparable) myArray.get(index * 2)).compareTo((Comparable) myArray.get(index * 2 + 1)) <= 0 && ((Comparable) myArray.get(index)).compareTo((Comparable) myArray.get(index * 2)) > 0)
        {
          // if left < right & check if top is greater than child
          myArray.set(index, myArray.get(index * 2));
          myArray.set(index * 2, item);
          index *= 2;
        }
        else if (((Comparable) myArray.get(index)).compareTo((Comparable) myArray.get(index * 2 + 1)) > 0)
        {
          // if left > right
          myArray.set(index, myArray.get(index * 2 + 1));
          myArray.set(index * 2 + 1, item);
          index = 2 * index + 1;
        }
        else
        {
          return;
        }
      }
      else if (2 * index < myArray.size() && ((Comparable) myArray.get(index * 2)).compareTo((Comparable) myArray.get(index)) < 0)
      {
        myArray.set(index, myArray.get(index * 2));
        myArray.set(index * 2, item);
        index *= 2;
      }
      else
      {
        return;
      }
    }
  }


  public boolean isEmpty()
  {
    return myArray.size() == 1;
  }


  public E peek()
  {
    if (isEmpty())
    {
      return null;
    }
    else
    {
      return myArray.get(1);
    }
  }


}
